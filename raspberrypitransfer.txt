void send_command(CClient& client, std::string cmd)
{
	std::string str;

	client.tx_str(cmd);
	std::cout << "\nClient Tx: " << cmd;

	if (cmd == "G 2")
	{
		cv::Mat im;

		if (client.rx_im(im) == true)
		{
			std::cout << "True" << std::endl;
			timeout_start = cv::getTickCount();
			if (im.empty() == false)
			{
				std::cout << "\nClient Rx: Image received";
				cv::imshow("rx", im);
				cv::waitKey(10);
			}
		}
		else
		{
			if ((cv::getTickCount() - timeout_start) / cv::getTickFrequency() > 1000)
			{
				// No response, disconnect and reconnect
				timeout_start = cv::getTickCount();
				client.close_socket();
				client.connect_socket(server_ip, server_port);
			}
		}
	}
	else
	{
		if (client.rx_str(str) == true)
		{
			timeout_start = cv::getTickCount();
			std::cout << "\nClient Rx: " << str;
		}
		else
		{
			if ((cv::getTickCount() - timeout_start) / cv::getTickFrequency() > 1000)
			{
				// No response, disconnect and reconnect
				timeout_start = cv::getTickCount();
				client.close_socket();
				client.connect_socket(server_ip, server_port);
			}
		}
	}
}

CMotor::CMotor(int leftPinPWM, int rightPinPWM) : _leftPinPWM(leftPinPWM), _rightPinPWM(rightPinPWM) {
    if (gpioInitialise() < 0) {
        std::cerr << "GPIO initialization failed" << std::endl;
    }
    gpioSetMode(_leftPinPWM, PI_OUTPUT);
    gpioSetMode(_rightPinPWM, PI_OUTPUT);
}

CMotor::~CMotor() {
    gpioTerminate();
}

void CMotor::forward(float speed, float duration) {
    gpioPWM(_leftPinPWM, speed * 255);
    gpioPWM(_rightPinPWM, speed * 255);
    std::this_thread::sleep_for(std::chrono::milliseconds(static_cast<int>(duration * 1000)));
    stop();
}

void CMotor::backward(float speed, float duration) {
    gpioPWM(_leftPinPWM, (1 - speed) * 127);
    gpioPWM(_rightPinPWM, (1 - speed) * 127);
    std::this_thread::sleep_for(std::chrono::milliseconds(static_cast<int>(duration * 1000)));
    stop();
}

void CMotor::left(float speed, float duration) {
    gpioPWM(_leftPinPWM, (1 - speed) * 127);
    gpioPWM(_rightPinPWM, speed * 255);
    std::this_thread::sleep_for(std::chrono::milliseconds(static_cast<int>(duration * 1000)));
    stop();
}
